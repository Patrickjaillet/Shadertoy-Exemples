#define GRILLE 20.
#define SEG_MAX 400
#define SEG_LIGNE 1

vec4 lire(ivec2 p) { return texelFetch(iChannel0, p, 0); }

float sd_boite(vec2 p, vec2 b) {
    vec2 d = abs(p) - b;
    return length(max(d, 0.0)) + min(max(d.x, d.y), 0.0);
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    vec2 aspect = iResolution.xy / iResolution.y;
    vec2 uv = fragCoord / iResolution.y;
    vec2 uv_centrer = uv - aspect * 0.5;
    
    vec4 h0 = lire(ivec2(0, 0));
    vec4 h1 = lire(ivec2(1, 0));
    vec4 h2 = lire(ivec2(2, 0));

    vec2 tete = h0.xy;
    float mort = h1.y;
    int longueur = int(h1.z);
    vec2 pomme = h2.xy;

    vec2 p_grille = (uv - (aspect - 1.0) * 0.5) * GRILLE;
    ivec2 cellule = ivec2(floor(p_grille));
    vec2 f_grille = fract(p_grille);

    vec3 couleur = vec3(0.08, 0.1, 0.12);
    
    if (p_grille.x >= 0.0 && p_grille.x < GRILLE && p_grille.y >= 0.0 && p_grille.y < GRILLE) {
        bool damier = (cellule.x + cellule.y) % 2 == 0;
        couleur = damier ? vec3(0.12, 0.15, 0.18) : vec3(0.1, 0.13, 0.16);
        
        vec2 bords = smoothstep(0.48, 0.5, abs(f_grille - 0.5));
        couleur *= 1.0 - (bords.x + bords.y) * 0.2;
        
        float dist_corps = 1e10;
        for (int i = 0; i < SEG_MAX; i++) {
            if (i >= longueur) break;
            vec4 segment = lire(ivec2(i, SEG_LIGNE));
            if (segment.z > 0.5) {
                float d = sd_boite(p_grille - (segment.xy + 0.5), vec2(0.42));
                dist_corps = min(dist_corps, d);
            }
        }
        
        float masque_serpent = smoothstep(0.02, 0.0, dist_corps);
        vec3 col_serpent = mix(vec3(0.1, 0.7, 0.2), vec3(0.3, 1.0, 0.5), exp(-dist_corps * 2.0));
        
        if (cellule == ivec2(tete)) {
            col_serpent = vec3(0.5, 1.0, 0.7);
        }
        
        couleur = mix(couleur, col_serpent, masque_serpent);

        float dist_pomme = length(p_grille - (pomme + 0.5)) - 0.35;
        float masque_pomme = smoothstep(0.02, 0.0, dist_pomme);
        vec3 col_pomme = vec3(1.0, 0.2, 0.1) * (1.0 + 0.3 * sin(iTime * 10.0));
        couleur = mix(couleur, col_pomme, masque_pomme);
        couleur += col_pomme * exp(-dist_pomme * 5.0) * 0.2;
    }

    if (mort > 0.5) {
        couleur = mix(couleur, vec3(0.2, 0.05, 0.05), 0.7);
        float flash = exp(-3.0 * length(uv - aspect * 0.5));
        couleur += vec3(1.0, 0.1, 0.0) * flash * abs(sin(iTime * 5.0));
    }

    couleur = pow(couleur, vec3(0.8));
    couleur *= 1.2 - 0.4 * length(uv - aspect * 0.5);

    fragColor = vec4(couleur, 1.0);
}

#define GRILLE 20
#define SEG_MAX 400
#define SEG_LIGNE 1
#define VITESSE 7.0

vec4 lire(ivec2 p) { return texelFetch(iChannel0, p, 0); }
bool touche(int k) { return texelFetch(iChannel1, ivec2(k, 0), 0).x > 0.5; }

float hachage(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    ivec2 p = ivec2(fragCoord);
    
    if (p.y > SEG_LIGNE || p.x > SEG_MAX) discard;

    vec4 h0 = lire(ivec2(0, 0));
    vec4 h1 = lire(ivec2(1, 0));
    vec4 h2 = lire(ivec2(2, 0));

    vec2 tete = h0.xy;
    int dir = int(h0.z);
    float score = h0.w;
    float vivant = h1.x;
    float mort = h1.y;
    float chrono = h1.z;
    int taille = int(h1.w);
    vec2 pomme = h2.xy;

    if (iFrame == 0 || (mort > 0.5 && iMouse.z > 0.5)) {
        if (p.y == SEG_LIGNE) {
            if (p.x < 4) fragColor = vec4(float(GRILLE/2 - 3 + p.x), float(GRILLE/2), 1.0, 0.0);
            else fragColor = vec4(0);
        } 
        else if (p == ivec2(0,0)) fragColor = vec4(float(GRILLE/2), float(GRILLE/2), 0, 0);
        else if (p == ivec2(1,0)) fragColor = vec4(0, 0, 0, 4);
        else if (p == ivec2(2,0)) fragColor = vec4(15, 10, 0, 0);
        else fragColor = vec4(0);
        return;
    }

    int n_dir = -1;
    if (touche(39) || touche(68)) n_dir = 0;
    if (touche(38) || touche(87) || touche(90)) n_dir = 1;
    if (touche(37) || touche(65) || touche(81)) n_dir = 2;
    if (touche(40) || touche(83)) n_dir = 3;

    if (n_dir >= 0 && n_dir != (dir + 2) % 4) dir = n_dir;
    if (vivant < 0.5 && n_dir >= 0) vivant = 1.0;

    bool avance = false;
    if (vivant > 0.5 && mort < 0.5) {
        chrono += 1.0;
        if (mod(chrono, VITESSE) < 1.0) {
            avance = true;
            vec2 deplacements[4];
            deplacements[0] = vec2(1, 0); deplacements[1] = vec2(0, 1);
            deplacements[2] = vec2(-1, 0); deplacements[3] = vec2(0, -1);
            
            vec2 futur = tete + deplacements[dir];

            if (futur.x < 0. || futur.x >= float(GRILLE) || futur.y < 0. || futur.y >= float(GRILLE)) {
                mort = 1.0;
            } else {
                for(int i = 0; i < SEG_MAX; i++) {
                    if (i >= taille) break;
                    if (futur == lire(ivec2(i, SEG_LIGNE)).xy) {
                        mort = 1.0;
                        break;
                    }
                }
            }

            if (mort < 0.5) {
                if (futur == pomme) {
                    score++;
                    taille = min(taille + 1, SEG_MAX);
                    pomme = floor(vec2(hachage(futur + iTime), hachage(futur.yx + iTime)) * float(GRILLE));
                }
                tete = futur;
            }
        }
    }

    if (p.y == SEG_LIGNE) {
        if (avance && mort < 0.5) {
            if (p.x == 0) fragColor = vec4(tete, 1.0, 0.0);
            else if (p.x < taille) {
                fragColor = lire(ivec2(p.x - 1, SEG_LIGNE));
                fragColor.z = 1.0;
            } else fragColor = vec4(0);
        } else {
            fragColor = lire(p); 
        }
    } 
    else if (p == ivec2(0,0)) fragColor = vec4(tete, float(dir), score);
    else if (p == ivec2(1,0)) fragColor = vec4(vivant, mort, chrono, float(taille));
    else if (p == ivec2(2,0)) fragColor = vec4(pomme, 0, 0);
    else fragColor = lire(p);
}
