float smin(float a, float b, float k) {
    float h = clamp(0.5 + 0.5 * (b - a) / k, 0.0, 1.0);
    return mix(b, a, h) - k * h * (1.0 - h);
}

mat2 rot(float a) {
    float s = sin(a), c = cos(a);
    return mat2(c, -s, s, c);
}

float sdBoite(vec3 p, vec3 b) {
    vec3 q = abs(p) - b;
    return length(max(q, 0.0)) + min(max(q.x, max(q.y, q.z)), 0.0);
}

float sdTore(vec3 p, vec2 t) {
    vec2 q = vec2(length(p.xz) - t.x, p.y);
    return length(q) - t.y;
}

float sdMandelbulb(vec3 p) {
    vec3 w = p;
    float m = dot(w, w);
    float dz = 1.0;
    for (int i = 0; i < 3; i++) {
        dz = 8.0 * pow(m, 3.5) * dz + 1.0;
        float r = length(w);
        float b = 8.0 * acos(w.y / r);
        float a = 8.0 * atan(w.x, w.z);
        w = p + pow(r, 8.0) * vec3(sin(b) * sin(a), cos(b), sin(b) * cos(a));
        m = dot(w, w);
        if (m > 4.0) break;
    }
    return 0.25 * log(m) * sqrt(m) / dz;
}

float sdMandelbox(vec3 p) {
    vec4 v = vec4(p, 1.0), v0 = v;
    for (int i = 0; i < 4; i++) {
        v.xyz = clamp(v.xyz, -1.0, 1.0) * 2.0 - v.xyz;
        float r2 = dot(v.xyz, v.xyz);
        if (r2 < 0.25) v *= 4.0; else if (r2 < 1.0) v /= r2;
        v = v * 2.5 / 1.0 + v0;
    }
    return (length(v.xyz) - 1.5) / v.w;
}

float sdKIFS(vec3 p) {
    float s = 1.0;
    for(int i=0; i<3; i++) {
        p = abs(p) - 0.5;
        p.xy *= rot(0.5);
        p.yz *= rot(0.8);
        p *= 1.5;
        s *= 1.5;
    }
    return length(p) / s;
}

float sdGyroide(vec3 p) {
    p *= 2.0;
    return abs(dot(sin(p), cos(p.yzx))) / 2.0 - 0.05;
}

float carte(vec3 p) {
    p.xy *= rot(iTime * 0.3);
    p.yz *= rot(iTime * 0.2);
    float t = mod(iTime, 60.0) / 4.0;
    int id = int(floor(t));
    float f = smoothstep(0.0, 1.0, fract(t));
    float d[15];
    d[0] = smin(length(p-0.4)-0.5, length(p+0.4)-0.5, 0.4);
    d[1] = sdMandelbox(p);
    d[2] = sdTore(p, vec2(0.8, 0.3));
    d[3] = sdMandelbulb(p);
    d[4] = sdBoite(p, vec3(0.7));
    d[5] = sdKIFS(p);
    d[6] = length(p) - 0.8;
    d[7] = max(sdBoite(p, vec3(0.8)), -sdGyroide(p));
    d[8] = sdMandelbox(p * 0.5) * 2.0;
    d[9] = length(p - clamp(p, -0.5, 0.5)) - 0.2;
    d[10] = sdGyroide(p * 2.0) * 0.5;
    d[11] = sdTore(p.zxy, vec2(0.6, 0.2));
    d[12] = sdKIFS(p * 0.8) / 0.8;
    d[13] = sdMandelbulb(p * 1.2) / 1.2;
    d[14] = smin(sdBoite(p, vec3(0.5)), length(p)-0.7, 0.2);
    return mix(d[id % 15], d[(id + 1) % 15], f);
}

void mainImage(out vec4 fCol, in vec2 fCoor) {
    vec2 uv = (fCoor - 0.5 * iResolution.xy) / iResolution.y;
    vec2 st = fCoor / iResolution.xy;
    vec3 ro = vec3(0, 0, -3), rd = normalize(vec3(uv, 1.2));
    float t = 0.0, d = 0.0, glo = 0.0;
    for (int i = 0; i < 70; i++) {
        d = carte(ro + rd * t);
        if (d < 0.002 || t > 8.0) break;
        t += d;
        glo += 0.012 / (0.015 + abs(d)); 
    }
    vec3 objCol = mix(vec3(0.1, 0.2, 0.5), vec3(0.9, 0.6, 0.8), 0.5 + 0.5 * sin(iTime * 0.5 + (ro+rd*t).z));
    vec3 col;
    if (t < 8.0) {
        vec3 p = ro + rd * t;
        vec2 e = vec2(0.015, 0.0);
        vec3 n = normalize(vec3(carte(p+e.xyy)-carte(p-e.xyy), carte(p+e.yxy)-carte(p-e.yxy), carte(p+e.yyx)-carte(p-e.yyx)));
        float dif = max(dot(n, normalize(vec3(1, 2, -1))), 0.0);
        float fre = pow(1.0 - max(dot(n, -rd), 0.0), 3.0);
        col = objCol * (dif + 0.3);
        col += fre * 0.6;
        vec2 refr = n.xy * 0.05;
        vec3 bg = texture(iChannel0, st + refr).rgb;
        col = mix(col, col + bg * fre, fre);
    } else {
        col = texture(iChannel0, st).rgb;
    }
    col += objCol * glo * 0.12;
    col = 1.0 - exp(-col * 1.2);
    fCol = vec4(col, 1.0);
}

#define T iTime * 15.0
#define N normalize
#define R(a) mat2(cos(a), -sin(a), sin(a), cos(a))
#define P(z) (vec3(cos((z)*0.02)*16.0, cos((z)*0.01)*32.0, (z)))

void mainImage(out vec4 fCol, vec2 fCoor) {
    float i, s, n, d, t = iTime;
    vec3 res = iResolution;

    vec2 u = (fCoor + fCoor - res.xy) / res.y;
    u += cos(t * vec2(0.5, 0.4)) * 0.5;

    float cycle = sin(t * 0.1) * 0.5 + 0.5;

    vec3 p = P(T * 4.0), ro = p,
         Z = N(P(T * 4.0 + 6.0) - p),
         X = N(vec3(Z.z, 0.0, -Z.x)),
         D = N(vec3(R(sin(T * 0.1) * 0.2) * u, 1.0) * mat3(-X, cross(X, Z), Z));    

    D.xy *= R(t * 0.3);
    D.xz *= R(t * 0.15);

    vec4 col = vec4(0.0);
    for(i = 0.0; i < 50.0 && d < 150.0; i++) {
        s = 0.2 + 0.1 * abs(s);
        d += s;
        p = ro + D * d;
        p.xy *= R(p.z * 0.014 + t * 0.2);
        
        s = mix(40.0, 28.0, cycle) - abs(p.y);
        
        for (n = 0.01; n < 3.0; n *= 3.0) {
            p *= 0.85;
            float pouls = sin(t * 0.4 + p.z * 0.1);
            s -= abs(dot(sin(0.3 * p.z + (3.0 + pouls) * t + p / n), vec3(n + n)));
        }
        
        col += (1.2 / s) * (0.7 + 0.3 * cos(p.z * 0.02 + vec4(0.0, 2.0, 4.0, 0.0)));
    }

    col = tanh(col * col / 180000.0 + 0.1 * dot(u, u));
    fCol = col;
}
