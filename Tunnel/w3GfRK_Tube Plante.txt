#define AA 1
#define ZERO (min(iFrame,0))

mat2 rotation(float a) {
    float c = cos(a), s = sin(a);
    return mat2(c, -s, s, c);
}

vec2 chemin(float z) {
    return vec2(sin(z * 0.2) * 1.5, cos(z * 0.15) * 1.2);
}

float minLissage(float a, float b, float k) {
    float h = clamp(0.5 + 0.5 * (b - a) / k, 0.0, 1.0);
    return mix(b, a, h) - k * h * (1.0 - h);
}

float carte(vec3 p) {
    vec3 q = p;
    q.xy -= chemin(q.z);
    
    float ang = atan(q.y, q.x);
    float ray = length(q.xy);
    
    float base = 2.5 - ray + 0.3 * sin(q.z * 1.5 - iTime * 1.5) * cos(ang * 6.0 + q.z);
    
    vec3 op = q;
    op.z = mod(op.z + iTime, 3.0) - 1.5;
    float nerv = length(vec2(ray - 2.6, op.z)) - 0.3;
    nerv -= 0.15 * sin(ang * 8.0 + p.z * 2.0);
    
    float d = minLissage(base, nerv, 0.6);
    d -= 0.03 * sin(p.x * 12.0) * sin(p.y * 12.0) * sin(p.z * 12.0);
    
    return d * 0.5;
}

float lancer(vec3 ro, vec3 rd, out float lueur) {
    float t = 0.01;
    lueur = 0.0;
    for (int i = 0; i < 100; i++) {
        float h = carte(ro + rd * t);
        lueur += 0.01 / (0.02 + abs(h));
        if (abs(h) < 0.002 * t || t > 40.0) return (t > 40.0) ? -1.0 : t;
        t += h;
    }
    return -1.0;
}

vec3 normale(vec3 p) {
    vec2 e = vec2(0.005, 0.0);
    return normalize(vec3(carte(p+e.xyy)-carte(p-e.xyy),
                          carte(p+e.yxy)-carte(p-e.yxy),
                          carte(p+e.yyx)-carte(p-e.yyx)));
}

float occlusion(vec3 p, vec3 n) {
    float occ = 0.0;
    float sca = 1.0;
    for (int i = ZERO; i < 4; i++) {
        float h = 0.06 + 0.12 * float(i);
        float d = carte(p + h * n);
        occ += (h - d) * sca;
        sca *= 0.9;
    }
    return clamp(1.0 - 2.0 * occ, 0.0, 1.0);
}

vec3 rendu(vec3 ro, vec3 rd) {
    vec3 col = vec3(0.01, 0.03, 0.02);
    float lueur = 0.0;
    float t = lancer(ro, rd, lueur);
    
    if (t > 0.0) {
        vec3 p = ro + rd * t;
        vec3 n = normale(p);
        float occ = occlusion(p, n);
        
        vec3 alb = vec3(0.15, 0.3, 0.15);
        alb += 0.08 * sin(p.z * 3.0) * vec3(0.2, 0.4, 0.1);
        
        vec3 ld = normalize(vec3(0.5, 0.5, 1.0));
        float diff = max(dot(n, ld), 0.0);
        float fres = pow(clamp(1.0 + dot(n, rd), 0.0, 1.0), 3.0);
        
        vec3 ecl = (diff * vec3(0.6, 0.8, 0.5) + 0.1 + fres * vec3(0.3, 0.7, 0.3)) * occ;
        col = alb * ecl;
        
        float brouil = 1.0 - exp(-0.02 * t * t);
        col = mix(col, vec3(0.04, 0.08, 0.06), brouil);
        col += vec3(0.2, 0.5, 0.1) * exp(-0.1 * t) * (1.0 - brouil);
    }
    
    col += vec3(0.2, 0.5, 0.1) * lueur * 0.015;
    return col;
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    vec2 p = (2.0 * fragCoord - iResolution.xy) / iResolution.y;
    float tc = iTime * 1.8;
    
    vec3 ro = vec3(chemin(tc), tc);
    vec3 ta = vec3(chemin(tc + 1.5), tc + 1.5);
    
    vec3 cw = normalize(ta - ro);
    vec3 cp = vec3(sin(iTime * 0.4), cos(iTime * 0.4), 0.0);
    vec3 cu = normalize(cross(cw, cp));
    vec3 cv = cross(cu, cw);
    vec3 rd = mat3(cu, cv, cw) * normalize(vec3(p, 1.4));

    vec3 col = rendu(ro, rd);
    col = pow(col, vec3(0.4545));
    
    vec2 q = fragCoord / iResolution.xy;
    col *= 0.5 + 0.5 * pow(16.0 * q.x * q.y * (1.0 - q.x) * (1.0 - q.y), 0.25);
    
    fragColor = vec4(col, 1.0);
}
